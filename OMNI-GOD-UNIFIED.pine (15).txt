f_sigmoid(float x, float steepness) =>
    1.0 / (1.0 + math.exp(-steepness * x))

// Map a layout input to a Pine position constant
f_pos(string p) =>
    p == "top_left" ? position.top_left : p == "top_right" ? position.top_right : p == "bottom_left" ? position.bottom_left : position.bottom_right

// Hurst-driven elastic EMA length — pure INT math (ta.ema needs int length)
// Trending (high Hurst) → short, responsive EMA · Choppy (low Hurst) → long, smooth EMA
f_elasticLen(float hVal) =>
    int el = useElastic ? (hVal > 0.66 ? minElastic : hVal > 0.34 ? (minElastic + maxElastic) / 2 : maxElastic) : fastLen
    el

// Multi-TF bias payload
f_biasData() =>
    float bE50  = ta.ema(close, 50)
    float bE200 = ta.ema(close, 200)
    bool  bBull = close > bE50 and bE50 > bE200
    bool  bBear = close < bE50 and bE50 < bE200
    float bDist = (close - bE50) / math.max(bE50, 0.001) * 100.0
    [bBull, bBear, bDist]

// HTF impulse / order block payload
f_obData() =>
    float oAtr  = ta.rma(ta.tr, atrLen)
    bool  oBull = (close - open) > oAtr * 1.5 and close > high[1] and close[1] < open[1]
    bool  oBear = (open - close) > oAtr * 1.5 and close < low[1] and close[1] > open[1]
    [oBull, oBear, high[1], low[1]]

// Macro payload
f_macroData() =>
    float mC    = close
    float mE50  = ta.ema(close, 50)
    float mE200 = ta.ema(close, 200)
    [mC, mE50, mE200]

// Radar row helper
f_radar_row(table tbl, int row, string factor, string status, bool isPass) =>
    table.cell(tbl, 0, row, factor,  bgcolor=color.rgb(10, 10, 10), text_color=color.gray,  text_size=size.small)
    table.cell(tbl, 1, row, status,  bgcolor=isPass ? color.new(color.green, 75) : color.new(color.red, 75), text_color=color.white, text_size=size.small)

// ── Head & Shoulders state machine — pivot-confirmed, non-repainting, PURE (no drawing — callable in security) ──
// state: 0 none · 1 LS+HEAD confirmed · 2 complete (neckline watched) · 3 neckline broken
f_hs_top() =>
    var int   tState = 0
    var float tLS = na
    var float tHead = na
    var float tRS = na
    var int   tLSbar = 0
    var int   tHeadBar = 0
    var int   tRSbar = 0
    var float tNL1 = na
    var int   tNL1bar = 0
    var float tNL2 = na
    var int   tNL2bar = 0
    var int   tBrokeBar = 0
    var array<float> phV = array.new_float()
    var array<int>   phB = array.new_int()

    float ph = ta.pivothigh(high, swingLen, swingLen)
    if not na(ph)
        array.unshift(phV, ph)
        array.unshift(phB, bar_index - swingLen)
        if array.size(phV) > 4
            array.pop(phV)
            array.pop(phB)

    float nlNow = na
    bool lsNow = false
    bool rsNow = false
    bool brkNow = false
    bool rstNow = false

    // Invalidation & expiry
    bool resetNow = false
    if tState == 1
        if not na(ph) and ph > tHead
            resetNow := true
        else if bar_index - tHeadBar > 150
            resetNow := true
    if tState == 2
        if close > tHead
            resetNow := true
        else if bar_index - tRSbar > 120
            resetNow := true
    if tState == 3 and bar_index - tBrokeBar > 10
        resetNow := true
    if resetNow
        tState := 0
        tLS := na
        tHead := na
        tRS := na
        tNL1 := na
        tNL2 := na
        tBrokeBar := 0
        rstNow := true

    // 0 → 1 : new head forms above a prior swing (left shoulder)
    if tState == 0 and array.size(phV) >= 2
        float headC = array.get(phV, 0)
        float lsC   = array.get(phV, 1)
        if headC > lsC + atr * 0.5
            tState := 1
            tLS := lsC
            tLSbar := array.get(phB, 1)
            tHead := headC
            tHeadBar := array.get(phB, 0)
            int kA = bar_index - tHeadBar
            int kB = bar_index - tLSbar
            tNL1 := low[kA]
            tNL1bar := tHeadBar
            for kk = kA + 1 to kB - 1
                if low[kk] < tNL1
                    tNL1 := low[kk]
                    tNL1bar := bar_index - kk
            lsNow := true

    // 1 → 2 : right shoulder below head, close to left shoulder
    if tState == 1 and not na(ph) and ph < tHead - atr * 0.5
        if math.abs(ph - tLS) <= atr * hsAtrTol
            tState := 2
            tRS := ph
            tRSbar := bar_index - swingLen
            int kA2 = bar_index - tRSbar
            int kB2 = bar_index - tHeadBar
            tNL2 := low[kA2]
            tNL2bar := tRSbar
            for kk = kA2 + 1 to kB2 - 1
                if low[kk] < tNL2
                    tNL2 := low[kk]
                    tNL2bar := bar_index - kk
            rsNow := true

    // 2 / 3 : live neckline + break watch
    if tState == 2 or tState == 3
        float nlSlope = (tNL2 - tNL1) / math.max(tNL2bar - tNL1bar, 1)
        nlNow := tNL2 + nlSlope * (bar_index - tNL2bar)
        if tState == 2 and close < nlNow
            tState := 3
            tBrokeBar := bar_index
            brkNow := true

    [tState, nlNow, tBrokeBar, tLS, tHead, tRS, tLSbar, tHeadBar, tRSbar, tNL1, tNL1bar, lsNow, rsNow, brkNow, rstNow]

            tier := 4
        if tier > maxLiqTier
            tier := maxLiqTier
    tier

// Losing modes get stricter entry thresholds; modes <35% WR after 8+ trades are bypassed
var int    tradeMode  = 0
var array<float> modePerf = array.new<float>(12)   // per mode: [trades, wins]
f_modePerfAdd(int m, bool win) =>
    int base = math.min(5, math.max(0, m)) * 2
    array.set(modePerf, base, array.get(modePerf, base) + 1.0)
    array.set(modePerf, base + 1, array.get(modePerf, base + 1) + (win ? 1.0 : 0.0))
f_modePerfN(int m) =>
    array.get(modePerf, math.min(5, math.max(0, m)) * 2)
f_modePerfWr(int m) =>
    float n = f_modePerfN(m)
    n >= 5 ? array.get(modePerf, math.min(5, math.max(0, m)) * 2 + 1) / n : 0.5
f_modeThreshAdj(int m) =>
    f_modePerfN(m) >= 5 ? math.max(-10.0, math.min(10.0, (0.5 - f_modePerfWr(m)) * 40.0)) : 0.0
f_modeViable(int m) =>
    float n = f_modePerfN(m)
    n < 8 or f_modePerfWr(m) >= 0.35

// HYBRID SELF-OPT — v10 ASOC rolling results + HYBRID v9 step-tuner, merged.
// Every asocWindow closed trades: WR < 45% → threshold +optStep (stricter);
// WR > 70% → threshold -optStep (looser). Offset persists until another sample flips it.
f_record_trade(bool isWin) =>
    f_modePerfAdd(tradeMode, isWin)
    if useASOC
        array.unshift(asocResults, isWin)
        if array.size(asocResults) > asocWindow
            array.pop(asocResults)
        float off = array.get(adaptState, 0)
        float wins = array.get(adaptState, 1)
        float trades = array.get(adaptState, 2)
        trades := trades + 1.0
        if isWin
            wins := wins + 1.0
        if trades >= asocWindow
            float wRate = wins / trades * 100.0
            if wRate < 45.0
                off := math.min(off + optStep, 10.0)
            else if wRate > 70.0
                off := math.max(off - optStep, -10.0)
            trades := 0.0
            wins := 0.0
        array.set(adaptState, 0, off)
        array.set(adaptState, 1, wins)
        array.set(adaptState, 2, trades)

for i = 0 to array.size(asocResults) - 1

// Least-squares fit over the last `cnt` pivots. x = bars since oldest fit pivot.
// Returns [slope per bar, value at oldest pivot, mean |residual|]
f_fitLine(array<float> P, array<int> B, int cnt) =>
    int start = array.size(P) - cnt
    float sx = 0.0
    float sy = 0.0
    float sxy = 0.0
    float sxx = 0.0
    int x0 = array.get(B, start)
    for i = start to array.size(P) - 1
        float x = array.get(B, i) - x0
        float y = array.get(P, i)
        sx += x
        sy += y
        sxy += x * y
        sxx += x * x
    float slope = 0.0
    float b0 = 0.0
    float denom = cnt * sxx - sx * sx
    if denom > 0
        slope := (cnt * sxy - sx * sy) / denom
        b0 := (sy - slope * sx) / cnt
    float resid = 0.0
    for i = start to array.size(P) - 1
        float x = array.get(B, i) - x0
        resid += math.abs(array.get(P, i) - (slope * x + b0))
    [slope, b0, resid / cnt]

        w += 3

            _pdH := line.new(bar_index - 30, pdHigh, bar_index + 15, pdHigh, color=color.new(color.orange, 35), style=line.style_dashed, width=1)
        if not na(pdLow)
            _pdL := line.new(bar_index - 30, pdLow, bar_index + 15, pdLow, color=color.new(color.orange, 35), style=line.style_dashed, width=1)
        if not na(pwHigh)
            _pwH := line.new(bar_index - 30, pwHigh, bar_index + 15, pwHigh, color=color.new(color.yellow, 50), style=line.style_dotted, width=1)
        if not na(pwLow)
            _pwL := line.new(bar_index - 30, pwLow, bar_index + 15, pwLow, color=color.new(color.yellow, 50), style=line.style_dotted, width=1)
        if not na(dailyPivot)
            _pvL := line.new(bar_index - 30, dailyPivot, bar_index + 15, dailyPivot, color=color.new(color.white, 38), style=line.style_dotted, width=2)
        if useWeekOpen and not na(weekOpen)
            _wkO := line.new(bar_index - 30, weekOpen, bar_index + 15, weekOpen, color=color.new(color.green, 35), style=line.style_dotted, width=1)

f_drawLevels()

f_drawLiqHeat() =>
    // Liquidity heat projection above/below last swings
    var box liqAboveBox = na
    var box liqBelowBox = na
    if showLiqHeat and barstate.islast

            box.delete(liqAboveBox)
        if not na(liqBelowBox)
            box.delete(liqBelowBox)
        if not na(lastPH)

        if not na(lastPL)
            liqBelowBox := box.new(bar_index - 5, lastPL - atr * 0.6, bar_index + 20, lastPL, bgcolor=color.new(color.green, 92), border_color=na)

f_drawLiqHeat()

f_drawNarrative() =>
    // ── NARRATIVE VOICE ──
    if showNarrative and barstate.islast
        string convStr = str.tostring(convNow, "#") + (convNow > nz(convNow[1]) ? " ▲" : convNow < nz(convNow[1]) ? " ▼" : " ·")
        string perfTag = ""
        if f_modePerfN(mode) >= 5
            perfTag := " " + str.tostring(f_modePerfWr(mode) * 100, "#") + "% WR (" + str.tostring(f_modePerfN(mode), "#") + ")"
        string narrative = "🧭 " + modeName + perfTag + " · " + symbolMode + " · Conv " + convStr
        if tradeState == 1
            string rStr = riskR > 0 ? str.tostring((close - entryPx) / riskR, "#.##") + "R" : "—"
            narrative := narrative + " · 🚀 LONG " + (close >= entryPx ? "+" : "") + rStr + " · " + (tp1Done ? (slPx > entryPx ? "BE + Profit Lock" : "BE Locked") : "En Route") + (convSoft ? " · ⚠ Soft Invalidation" : "") + (isCounterBias ? " · ⚡ C-Bias" : "") + " · " + catalyst
        else if tradeState == -1
            string rStr = riskR > 0 ? str.tostring((entryPx - close) / riskR, "#.##") + "R" : "—"
            narrative := narrative + " · 🚀 SHORT " + (close <= entryPx ? "+" : "") + rStr + " · " + (tp1Done ? (slPx < entryPx ? "BE + Profit Lock" : "BE Locked") : "En Route") + (convSoft ? " · ⚠ Soft Invalidation" : "") + (isCounterBias ? " · ⚡ C-Bias" : "") + " · " + catalyst
        else
            string ctx = "Equilibrium"
            if cbHalt
                ctx := "⛔ CIRCUIT BREAKER — day done (" + str.tostring(dayR, "#.##") + "R), protect capital"
            else if cautionMode
                ctx := "⚠ " + (isNfpDay ? "NFP Day" : "London Fix")
            else if spikeBlocked
                ctx := "🛡 Spike Shield"
            else if vBotTrig
                ctx := "⚡ V-bottom " + str.tostring(nz(barsBot)) + "b ago"
            else if vTopTrig
                ctx := "⚡ V-top " + str.tostring(nz(barsTop)) + "b ago"
            else if pattern != PAT_NONE
                ctx := "📐 " + patternName + (not na(barsApex) ? " · apex ~" + str.tostring(barsApex, "#") + "b" : "")
            else if isWeekend
                ctx := "🌙 WEEKEND · thin flow"
            else if asiaBreakLong
                ctx := "🌏 ASIA RANGE BREAK ↑"
            else if asiaBreakShort
                ctx := "🌏 ASIA RANGE BREAK ↓"
            else if useHs and ((hsT_state == 3 and bar_index - hsT_broke <= 5) or (hsB_state == 3 and bar_index - hsB_broke <= 5))
                ctx := "⚠ H&S " + (hsT_state == 3 ? "TOP BROKEN" : "BOTTOM BROKEN")
            else if useTrapEngine and bullTrapActive and bearTrapActive
                ctx := "⚔ TRAPS BOTH SIDES — thin edge, stand down"
            else if useTrapEngine and bullTrapActive
                ctx := "🪤 BULL TRAP SPRUNG — fade the breakout"
            else if useTrapEngine and bearTrapActive
                ctx := "🪤 BEAR TRAP SPRUNG — fade the breakdown"
            else if useBeyondFvg and bfgState == 2
                ctx := "🌊 BEYOND-FVG reclaim ↑ — longs past the gap"
            else if useBeyondFvg and rfgState == 2
                ctx := "🌊 BEYOND-FVG reclaim ↓ — shorts past the gap"
            else if baitLong or baitShort
                ctx := baitLong ? "⚠ Inside bull FVG — bait zone, wait for the sweep" : "⚠ Inside bear FVG — bait zone, wait for the sweep"
            else if pendingBuy
                ctx := "🟢 BUY READY " + str.tostring(buyScore, "#") + "% · " + buyWhy
            else if pendingSell
                ctx := "🔴 SELL READY " + str.tostring(sellScore, "#") + "% · " + sellWhy
            else if squeeze30
                ctx := "🟡 Compression — breakout watch"
            else if regimeVolatile
                ctx := "🟣 Volatile expansion"
            else if inPremium
                ctx := "Premium zone — favor shorts"
            else if inDiscount
                ctx := "Discount zone — favor longs"
            narrative := narrative + " · " + ctx

        var label narLabel = na
        if not na(narLabel)
            label.delete(narLabel)
        narrative := "🧭 " + modeName + " · " + narrative
        narLabel := label.new(bar_index + 1, close, narrative, color=color.rgb(8, 8, 12), textcolor=color.white, style=label.style_label_left, size=size.small, textalign=text.align_left)

f_drawNarrative()
